namespace = er_timur

er_timur.1 = {
    type = country_event
    hidden = no
    title = er_timur.1.title
    desc = er_timur.1.desc

	fire_only_once = yes
	outcome = positive

	illustration_tags = {
        10 = regular
        10 = interior
    }

	dynamic_historical_event = {
		tag = TIM
		from = 1350.1.1
		to = 1550.1.1
		monthly_chance = 2
	}

    trigger = {
	
		NOT = { is_situation_active = situation:rise_of_timur }

		government_type ?= government_type:monarchy
		num_locations_owned_or_owned_by_subjects_or_below < 1000

		region:persia_region = {
			any_ownable_location_in_region = {
				owner = c:TIM
				count > 200
			}
		}
		
		NOT = { country_exists = c:IRA }

		location:hamadan = {
			owner = c:TIM
		}
    }

    option = {
        name = er_timur.1.a
		
		
		form_country = formable_country:IRA_f
		set_capital = location:hamadan

		ai_will_select = { value = 70 }
    }

	option = {
		name = er_timur.1.b

		ai_will_select = { value = 30 }
	}
}
