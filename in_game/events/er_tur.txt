	type = country_event

	desc = rise_of_the_ottomans.217.desc
	outcome = neutral
	image = "gfx/interface/illustrations/event/backgrounds/exterior/soldiers/syrian_soldiers_exterior.dds"

	immediate = {

			while =  {

				capital = { create_sub_unit_of_category = sub_unit_category:army_heavy_infantry }
			}
		}
	}

	option = {
		name = rise_of_the_ottomans.217.a

		scope:target_subject = {

			show_as_tooltip = {
				while =  {
					count = 2

			}
			
			change_culture = culture:turkish_culture
			set_court_language = dialect:turkish_dialect

				limit = { NOT = { religion.group = religion_group:muslim } }
