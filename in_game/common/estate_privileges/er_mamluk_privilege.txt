    estate = nobles_estate

    potential = {
        tag = MAM
    }

    # allow determines if it can be selected (empty/always true if they start with it)
    allow = {

    # Modifiers must be placed in country_modifier per the documentation
    country_modifier = {
        global_nobles_estate_power = 2
        nobles_estate_target_satisfaction = -0.1

        global_manpower_modifier = 0.1
        army_heavy_cavalry_maintenance_cost_modifier = -0.1
        army_light_cavalry_maintenance_cost_modifier = -0.1

        global_levy_size_modifier = -0.35
        nobles_estate_max_tax = -0.15
        land_morale_modifier = -0.25

    # Evaluates whether you are legally allowed to click the revoke button
    can_revoke = {
        # Check if the country asset matches or exceeds the threshold

    on_deactivate = {
        trigger_event_non_silently = { id = mamluk_balance.2 }
    }
}
