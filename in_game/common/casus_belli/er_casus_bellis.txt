cb_border_feuds = {
    # 1. Visibility: Everyone can see this CB option in the diplomacy UI
    create_visible = {

        current_year < 1500
        scope:target = {
            is_real_country = yes
        }
        
        NOT = { 
            scope:target = { 
                is_member_of_international_organization = international_organization:hre 
            } 
        }

        OR = {
            is_neighbor_of = scope:target
            any_subject = {
                NOT = { is_subject_type = tributary }
                is_neighbor_of = scope:target
            }
        }
    }

    create_enabled = {
    }

    declare_enabled = {
    }

    province = {
        # Any province owned by the target that borders you is valid to target
        owner = scope:recipient
        any_location_in_province = {
            is_neighbor_of = scope:actor
        }
    }

    war_goal_type = border_feuds
    can_expire = yes

    ai_will_do = { add = { value = 100 } }
}

### Russia

cb_unify_the_rus = {
    # 1. Visibility: Everyone can see this CB option in the diplomacy UI
    create_visible = {
        is_real_country = yes

        culture ?= {
            has_culture_group = culture_group:russian_group
        }
        
        always = yes
    }

    # 2. Creation: Can we actually generate this CB against the target?
    create_enabled = {
        # Check that the target isn't yourself
        NOT = { this = scope:target }
        current_date >= 1400.1.1

		culture ?= {
			has_culture_group = culture_group:russian_group
		}

        # 2. Defender (target country) belongs to the Russian culture group
        scope:target = {
            is_real_country = yes

            culture ?= {
                has_culture_group = culture_group:russian_group
            }
        }

        # 3. Prevent self-targeting or targeting your own vassals
        NOT = { scope:target = root }
        NOT = { scope:target = { is_subject_of = root } }
        
        OR = {
            is_neighbor_of = scope:target
            any_subject = {
                NOT = { is_subject_type = tributary }
                is_neighbor_of = scope:target
            }
        }


    }

    declare_enabled = {
    }
    
    ai_will_do = {
        add = { value = 1000 }
    }
    
	ai_subjugation_desire = -100
    
    war_goal_type = unification_of_russia
    can_expire = yes
}
